const ALLOWED_ORIGIN = 'https://dalbonz.github.io';

const PERIOD_LABELS = {
  '1d': '최근 1일',
  '1w': '최근 1주',
  '1mo': '최근 1개월',
  '3mo': '최근 3개월',
  '1y': '최근 1년'
};

const TOPIC_LABELS = {
  flow: '큰손 수급(기관·외국인 움직임)',
  technical: '기술적 흐름(추세·거래량·기술지표)',
  issue: '이슈 탐색(뉴스·공시 영향)',
  fundamental: '기업 성과(성장성과 수익성)',
  valuation: '가격 매력(가치·동종업계 비교)',
  risk: '리스크 점검(반대 근거까지 확인)'
};

const PERSPECTIVE_LABELS = {
  balanced: '균형분석',
  bull: '상승 시나리오 중심',
  bear: '하락 시나리오 중심',
  conservative: '보수적 분석',
  contrarian: '반대의견 제시',
  simple: '쉽게 설명'
};

const LENGTH_TOKENS = { short: 500, normal: 1000, deep: 2000 };

function corsHeaders() {
  return {
    'Access-Control-Allow-Origin': ALLOWED_ORIGIN,
    'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
    'Access-Control-Allow-Headers': 'content-type, x-app-key',
    'content-type': 'application/json'
  };
}

// 야후 파이낸스 시세 프록시 (무료 CORS 프록시 3종이 모두 죽어서 대체)
async function handleYfinance(url, env, request) {
  if (request.headers.get('X-App-Key') !== env.APP_SHARED_KEY) {
    return new Response(JSON.stringify({ error: 'unauthorized' }), { status: 401, headers: corsHeaders() });
  }
  const symbol = url.searchParams.get('symbol') || '';
  const interval = url.searchParams.get('interval') || '1d';
  const range = url.searchParams.get('range') || '1mo';
  if (!symbol || symbol.length > 20) {
    return new Response(JSON.stringify({ error: 'invalid_symbol' }), { status: 400, headers: corsHeaders() });
  }
  const yahooUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(symbol)}?interval=${encodeURIComponent(interval)}&range=${encodeURIComponent(range)}`;
  try {
    const res = await fetch(yahooUrl, { headers: { 'User-Agent': 'Mozilla/5.0' } });
    const text = await res.text();
    return new Response(text, { status: res.status, headers: corsHeaders() });
  } catch (e) {
    return new Response(JSON.stringify({ error: 'fetch_failed' }), { status: 502, headers: corsHeaders() });
  }
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    if (request.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders() });
    }
    if (url.pathname === '/yfinance' && request.method === 'GET') {
      return handleYfinance(url, env, request);
    }
    if (request.method !== 'POST') {
      return new Response('Method Not Allowed', { status: 405, headers: corsHeaders() });
    }
    if (request.headers.get('X-App-Key') !== env.APP_SHARED_KEY) {
      return new Response(JSON.stringify({ error: 'unauthorized' }), { status: 401, headers: corsHeaders() });
    }

    let body;
    try {
      body = await request.json();
    } catch (e) {
      return new Response(JSON.stringify({ error: 'invalid_json' }), { status: 400, headers: corsHeaders() });
    }

    const { name, symbol, period, topics, perspective, length, stats, series } = body || {};
    if (!symbol || typeof symbol !== 'string' || symbol.length > 20) {
      return new Response(JSON.stringify({ error: 'invalid_symbol' }), { status: 400, headers: corsHeaders() });
    }

    const periodLabel = PERIOD_LABELS[period] || PERIOD_LABELS['1mo'];
    const topicKeys = (Array.isArray(topics) ? topics : []).filter(t => TOPIC_LABELS[t]);
    if (!topicKeys.length) topicKeys.push('technical');
    const topicLabels = topicKeys.map(t => TOPIC_LABELS[t]);
    const perspectiveLabel = PERSPECTIVE_LABELS[perspective] || PERSPECTIVE_LABELS['balanced'];
    const maxTokens = LENGTH_TOKENS[length] || LENGTH_TOKENS['normal'];

    const safeName = String(name || symbol).slice(0, 40);
    const NO_DATA_TOPICS = new Set(['issue', 'fundamental', 'valuation']);
    const hasNoDataTopic = topicKeys.some(t => NO_DATA_TOPICS.has(t));

    let dataSection = '실제 시세 데이터가 전달되지 않았습니다. 구체적 수치는 언급하지 말고 일반적인 맥락만 설명하세요.';
    if (stats && typeof stats === 'object' && Array.isArray(series)) {
      const safeSeries = series
        .filter(p => p && typeof p.t === 'string' && typeof p.c === 'number' && isFinite(p.c))
        .slice(0, 60)
        .map(p => `${p.t.slice(0, 20)}: ${p.c}`)
        .join('\n');
      const s = stats;
      const num = v => (typeof v === 'number' && isFinite(v)) ? v : null;
      dataSection = `기간 시작가: ${num(s.firstClose) ?? '확인불가'}
기간 종가: ${num(s.lastClose) ?? '확인불가'}
기간 고가: ${num(s.periodHigh) ?? '확인불가'}
기간 저가: ${num(s.periodLow) ?? '확인불가'}
기간 등락률: ${num(s.periodPct) ?? '확인불가'}%
기간 거래량 합계: ${num(s.periodVol) ?? '확인불가'}

시계열 샘플(날짜/시각: 종가):
${safeSeries || '(데이터 없음)'}`;
    }

    const prompt = `당신은 개인 투자자를 위한 주식 분석 브리핑을 작성하는 애널리스트입니다.

종목: ${safeName} (${symbol})
분석 기간: ${periodLabel}
분석 관점: ${perspectiveLabel}
확인할 항목:
${topicLabels.map(t => '- ' + t).join('\n')}

[실제 시세 데이터 — 아래 데이터 범위 안에서만 구체적 수치·날짜·추세를 언급하세요. 데이터에 없는 특정 날짜나 사건을 단정적으로 지어내지 마세요]
${dataSection}

위 항목을 중심으로 이 종목에 대한 브리핑을 작성하세요.${hasNoDataTopic ? ' "이슈 탐색", "기업 성과", "가격 매력" 항목은 실시간 뉴스·재무·밸류에이션 데이터가 제공되지 않았으므로, 해당 부분은 일반적/구조적 해석 위주로 작성하고 "실시간 데이터 기반이 아닌 일반적 해석" 임을 해당 섹션 앞에 짧게 밝히세요.' : ''} 투자 조언이 아닌 참고 정보임을 전제로 하되, 매 문단마다 disclaimer를 반복하지는 마세요.`;

    const anthropicRes = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'content-type': 'application/json',
        'x-api-key': env.ANTHROPIC_API_KEY,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-6',
        max_tokens: maxTokens,
        messages: [{ role: 'user', content: prompt }]
      })
    });

    if (!anthropicRes.ok) {
      const errText = await anthropicRes.text();
      return new Response(JSON.stringify({ error: 'anthropic_error', detail: errText }), { status: 502, headers: corsHeaders() });
    }

    const data = await anthropicRes.json();
    const text = (data.content || []).map(c => c.text || '').join('\n').trim();

    return new Response(JSON.stringify({ text }), { headers: corsHeaders() });
  }
};
